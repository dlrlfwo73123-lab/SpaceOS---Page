  4. 이 종합본 파일(SpaceOS 종합본.html)도 바탕화면에 함께 둡니다.

3. 지도 배경에 CARTO(basemaps.cartocdn.com)와 OSM 공식 타일(tile.openstreetmap.org)은 쓰지 않는다. CARTO는 2026년부터 API 키가 없으면 "API KEY REQUIRED" 워터마크가 뜨고, OSM은 로컬 파일에서 Referer 문제로 차단될 수 있다. 대신 아래 addBasemap 함수를 그대로 써서 OpenFreeMap(키 없음)을 깔고, 실패하면 Esri 타일로 자동 대체한다. 이 함수는 지도에 center와 zoom을 먼저 준 다음 호출해야 한다(L.map('map', {center, zoom}) → addBasemap(map)).
// 배경지도 — CARTO 기본 타일은 2026년부터 API 키가 없으면 워터마크가 떠서,
// 키 없이 쓸 수 있는 OpenFreeMap(벡터)을 쓰고 WebGL · CDN 문제로 못 띄우면 Esri 래스터 타일로 바꿉니다.
function addBasemap(map){
  let done = false;
  const fallback = () => {
    if (done) return; done = true;
    L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Street_Map/MapServer/tile/{z}/{y}/{x}', { maxZoom:20, maxNativeZoom:19, attribution:'Tiles &copy; Esri' }).addTo(map);
  };
  const webgl = (() => { try { const c = document.createElement('canvas'); return !!(c.getContext('webgl2') || c.getContext('webgl')); } catch (e) { return false; } })();
  if (!webgl || !window.maplibregl || !L.maplibreGL) return fallback();
  try {
    const gl = L.maplibreGL({ style:'https://tiles.openfreemap.org/styles/liberty', attributionControl:{ customAttribution:'<a href="https://openfreemap.org" target="_blank" rel="noopener">OpenFreeMap</a> &copy; <a href="https://www.openmaptiles.org/" target="_blank" rel="noopener">OpenMapTiles</a> &copy; <a href="https://www.openstreetmap.org/copyright" target="_blank" rel="noopener">OpenStreetMap</a>' } }).addTo(map);
    const m = gl.getMaplibreMap();
    const giveUp = () => { if (done) return; clearTimeout(timer); map.removeLayer(gl); fallback(); };
    const timer = setTimeout(giveUp, 8000);
    m.on('error', giveUp);
    m.on('load', () => {
      if (done) return;
      done = true; clearTimeout(timer);
      // 지명은 한국어 우선
      m.getStyle().layers.forEach(l => {
        const tf = l.layout && l.layout['text-field'];
        if (l.type === 'symbol' && tf && JSON.stringify(tf).includes('name')) m.setLayoutProperty(l.id, 'text-field', ['coalesce', ['get', 'name:ko'], ['get', 'name']]);
      });
    });
  } catch (e) { fallback(); }
}

// ── 층 배치 계산: 전용면적으로 호실 크기를 정한다 ─────────────────────────
// 앞줄(도로 쪽, 깊이 fd)은 왼쪽→오른쪽, 복도(corr) 뒤 뒷줄은 오른쪽→왼쪽으로 채운다.
// 코어(계단 · EV · WC)는 뒤쪽 왼쪽 모서리. 남는 자리는 공용(com)으로 표시한다.
function layoutFloor(b, f){
  const W = b.W, D = b.D, X0 = -W / 2, X1 = W / 2, Z0 = -D / 2, Z1 = D / 2;
  const cw = b.core.w, cd = b.core.d, corr = b.corr || 0;
  const fd = b.fd || (D - cd);
  const core = { kind:'core', label:'계단 · EV · WC', x0:X0, x1:X0 + cw, z0:Z0, z1:Z0 + cd };
  const out = { pieces:[core], units:{}, overflow:[], plate:W * D };
  const put = (un, kind, p, label) => {
    const no = un ? un.no : null, lab = label || (un && un.name);
    // 같은 호실(또는 같은 공용 칸)이 바로 옆에 이어지면 한 칸으로 합친다
    const last = out.pieces[out.pieces.length - 1];
    if (last && last.kind === kind && last.no === no && last.label === lab && Math.abs(last.z0 - p.z0) < 1e-6 && Math.abs(last.z1 - p.z1) < 1e-6
        && (Math.abs(last.x1 - p.x0) < 1e-6 || Math.abs(last.x0 - p.x1) < 1e-6)){
      last.x0 = Math.min(last.x0, p.x0); last.x1 = Math.max(last.x1, p.x1); return;
    }
    const pc = { kind, no, st:un ? un.st : 'com', label:lab, ...p };
    out.pieces.push(pc);
    if (un) (out.units[un.no] = out.units[un.no] || []).push(pc);
  };
  // 줄 하나 채우기: [z0,z1] 띠에서 코어와 겹치는 부분은 빼고, dir 방향으로 면적만큼 x를 잘라 준다
  const fillRow = (list, z0, z1, dir, restLabel) => {
    const coreCut = Math.max(0, Math.min(z1, Z0 + cd) - Math.max(z0, Z0));
    const segs = [[X0, X0 + cw, coreCut], [X0 + cw, X1, 0]];   // [x0, x1, 빼는 깊이]
    if (dir < 0) segs.reverse();
    let si = 0, cur = dir > 0 ? X0 : X1;
    const zr = cut => cut > 0 ? [Math.max(z0, Z0 + cd), z1] : [z0, z1];
    const take = (need, un, kind, label) => {
      while (need > 0.01 && si < segs.length){
        const [sx0, sx1, cut] = segs[si];
        const depth = (z1 - z0) - cut;
        const end = dir > 0 ? sx1 : sx0;
        const room = Math.abs(end - cur) * depth;
        if (depth <= 0.01 || room <= 0.01){ si++; cur = dir > 0 ? (segs[si] || [])[0] : (segs[si] || [])[1]; continue; }
        const used = Math.min(need, room), wdt = used / depth;
        const nx = cur + dir * wdt;
        const [a, b2] = zr(cut);
        put(un, kind, { x0:Math.min(cur, nx), x1:Math.max(cur, nx), z0:a, z1:b2 }, label);
        need -= used; cur = nx;
        if (Math.abs(cur - end) < 1e-6){ si++; if (si < segs.length) cur = dir > 0 ? segs[si][0] : segs[si][1]; }
      }
      return need;
    };
    list.forEach(un => {
      const left = take(un.a, un, un.st === 'com' ? 'com' : 'unit');
      if (left > 0.5) out.overflow.push({ no:un.no, missing:Math.round(left) });
    });
    // 남는 자리 → 공용
    let rest = 0;
    segs.forEach(([sx0, sx1, cut], i) => {
      if (i < si) return;
      const from = i === si ? cur : (dir > 0 ? sx0 : sx1);
      rest += Math.abs((dir > 0 ? sx1 : sx0) - from) * ((z1 - z0) - cut);
    });
    if (rest > 0.5) take(rest, null, 'com', restLabel);
  };
  const full = f.units.find(x => x.pos === 'full');
  if (full){
    fillRow([{ ...full, a: W * fd }], Z1 - fd, Z1, 1, '공용');
    if (corr) put(null, 'hall', { x0:X0, x1:X1, z0:Z1 - fd - corr, z1:Z1 - fd }, '복도 (공용)');
    fillRow([{ ...full, a: W * (D - fd - corr) - cw * cd }], Z0, Z1 - fd - corr, -1, '공용');
    return out;
  }
  fillRow(f.units.filter(x => (x.pos || 'front') === 'front'), Z1 - fd, Z1, 1, f.rest || '공용');
  if (corr) put(null, 'hall', { x0:X0, x1:X1, z0:Z1 - fd - corr, z1:Z1 - fd }, '복도 (공용)');
  if (Z1 - fd - corr - Z0 > 0.3) fillRow(f.units.filter(x => x.pos === 'back'), Z0, Z1 - fd - corr, -1, f.backRest || (corr ? '공용' : '복도 · 화장실'));
  return out;
}
const pieceArea = p => (p.x1 - p.x0) * (p.z1 - p.z0);

// ── 공실 집계: 같은 건물도 기준에 따라 공실률이 달라진다 ──────────────────
// 계약면적(분양면적) = 전용면적 + 그 층 공용(코어 · 복도 · 화장실)을 전용면적 비율로 나눈 몫
//                    + 건물 공용(로비 · 관리실 · 주차장)을 건물 전체 전용면적 비율로 나눈 몫
// 주거층(res)은 상가 공실 집계에서 뺀다.
const BLDG_COMMON = /로비|관리실|주차/;
function areaShares(b){
  const floors = b.floors.map(f => {
    const lay = layoutFloor(b, f);
    const leas = f.units.filter(x => LEASABLE.includes(x.st));
    const excl = leas.reduce((s, x) => s + x.a, 0);
    let floorPool = 0, bldPool = 0;
    if (!leas.length){
      if (f.units.every(x => x.st === 'park')) bldPool = lay.plate;
    } else {
      lay.pieces.forEach(p => {
        if (p.kind === 'unit') return;
        if (BLDG_COMMON.test(p.label || '')) bldPool += pieceArea(p); else floorPool += pieceArea(p);
      });
    }
    return { f, lay, excl, floorPool, bldPool };
  });
  const exclAll = floors.reduce((s, x) => s + x.excl, 0);
  const bldPool = floors.reduce((s, x) => s + x.bldPool, 0);
  const contractOf = (fx, a) => a + (fx.excl ? fx.floorPool * a / fx.excl : 0) + (exclAll ? bldPool * a / exclAll : 0);
  return { floors, exclAll, bldPool, contractOf };
}
function contractArea(b, lv, un){
  const sh = areaShares(b), fx = sh.floors.find(x => x.f.lv === lv);
  return fx ? sh.contractOf(fx, un.a) : un.a;
}
function vacancyStats(b){

  const rows = [];
  sh.floors.forEach(fx => fx.f.units.forEach(x => {
    if (LEASABLE.includes(x.st)) rows.push({ ...x, lv:fx.f.lv, use:fx.f.use, contract:sh.contractOf(fx, x.a) });
  }));
  const sum = (arr, k) => arr.reduce((s, x) => s + (typeof k === 'function' ? k(x) : x[k]), 0);
  const rate = (v, n) => n ? v / n * 100 : 0;
  const pack = (r, vf, k) => { const v = r.filter(vf); return { v:sum(v, k), n:sum(r, k), rate:rate(sum(v, k), sum(r, k)) }; };
  const rooms = x => x.rooms || 1, one = () => 1;
  const grp = pred => pack(rows.filter(pred), isVacant, 'a');
  const out = {
    rows,
    rooms:pack(rows, isVacant, rooms),
    stores:pack(rows, isVacant, one),
    area:pack(rows, isVacant, 'a'),
    contract:pack(rows, isVacant, 'contract'),
    actual:pack(rows, x => isVacant(x) || x.st === 'dark', 'a'),
    exUnsold:pack(rows.filter(x => x.st !== 'unsold'), isVacant, 'a'),
    byUse:{ '근생':grp(x => x.use === '근생'), '업무':grp(x => x.use === '업무') },

    unsold:sum(rows.filter(x => x.st === 'unsold'), 'a'), dark:sum(rows.filter(x => x.st === 'dark'), 'a'),
    merged:rows.filter(x => (x.rooms || 1) > 1),
    exclAll:sh.exclAll, bldPool:sh.bldPool,
    gfa3d:b.floors.length * b.W * b.D, gfaLedger:b.ledger.gfa,

  const rs = [out.rooms.rate, out.stores.rate, out.area.rate, out.contract.rate, out.actual.rate, out.exUnsold.rate];
  out.min = Math.min(...rs); out.max = Math.max(...rs);
  return out;
}

const DIR8 = ['북','북동','동','남동','남','남서','서','북서'];

// 건물 좌표(x 오른쪽, z 앞쪽) → 지도상 동 · 북 거리(m)
function localToEN(b, x, z){
  const r = Math.PI / 180, bx = (b.face - 90) * r, bz = b.face * r;
  return { e: x * Math.sin(bx) + z * Math.sin(bz), n: x * Math.cos(bx) + z * Math.cos(bz) };
}
// 호실이 건물 안 어디에 있는지 사람이 읽을 수 있는 말로
function unitWhere(b, f, no){
  const ps = layoutFloor(b, f).units[no]; if (!ps || !ps.length) return '';
  const a = ps.reduce((s, p) => s + pieceArea(p), 0);
  const cx = ps.reduce((s, p) => s + (p.x0 + p.x1) / 2 * pieceArea(p), 0) / a;
  const cz = ps.reduce((s, p) => s + (p.z0 + p.z1) / 2 * pieceArea(p), 0) / a;
  const side = cx < -b.W / 6 ? '왼쪽' : cx > b.W / 6 ? '오른쪽' : '가운데';
  const row = cz > 0 ? `앞줄(${b.road} 쪽, ${dirName(b.face)}향)` : `뒷줄(${dirName(b.face + 180)}향)`;
  return `${row} · 도로에서 보아 ${side}`;
}

}

function rgb2hsl([r, g, b]){ r /= 255; g /= 255; b /= 255; const mx = Math.max(r, g, b), mn = Math.min(r, g, b); let h = 0, s = 0; const l = (mx + mn) / 2; if (mx !== mn){ const d = mx - mn; s = l > .5 ? d / (2 - mx - mn) : d / (mx + mn); h = mx === r ? (g - b) / d + (g < b ? 6 : 0) : mx === g ? (b - r) / d + 2 : (r - g) / d + 4; h /= 6; } return [h * 360, s * 100, l * 100]; }
function hsl2rgb([h, s, l]){ h /= 360; s /= 100; l /= 100; if (!s) return [l, l, l].map(v => v * 255); const q = l < .5 ? l * (1 + s) : l + s - l * s, p = 2 * l - q; const f = t => { t = (t + 1) % 1; return t < 1/6 ? p + (q - p) * 6 * t : t < 1/2 ? q : t < 2/3 ? p + (q - p) * (2/3 - t) * 6 : p; }; return [f(h + 1/3), f(h), f(h - 1/3)].map(v => v * 255); }
const adjHex = (h, dl = 0, ds = 0) => { const [H, S, L] = rgb2hsl(hex2rgb(h)); return rgb2hex(hsl2rgb([H, Math.max(0, Math.min(100, S + ds)), Math.max(0, Math.min(100, L + dl))])); };

function fixContrast(bg, fg, target = 4.5){
  const step = (a, b) => { const dir = lum(b) > lum(a) ? -1 : 1; for (let i = 1; i <= 40; i++){ const c = adjHex(a, dir * i * 1.5); if (contrast(c, b) >= target) return [i, c]; } return [99, dir < 0 ? '#000000' : '#FFFFFF']; };

  return ib <= iff ? { part:'바탕', from:bg, to:cb } : { part:'글자', from:fg, to:cf };
}
// 색온도(K) → 빛 색 (Tanner Helland 근사)
function kelvinHex(K){
  const t = K / 100;
  const r = t <= 66 ? 255 : 329.698727446 * Math.pow(t - 60, -.1332047592);
  const g = t <= 66 ? 99.4708025861 * Math.log(t) - 161.1195681661 : 288.1221695283 * Math.pow(t - 60, -.0755148492);
  const b = t >= 66 ? 255 : t <= 19 ? 0 : 138.5177312231 * Math.log(t - 10) - 305.0447927307;
  return rgb2hex([r, g, b]);
}

지금 시안의 한계
  • 유동인구 · 매출 · 점포 위치 · 적합도 등 핵심 수치가 가상입니다. 성수 공실률 · 임대료, 수수료, 인테리어 단가만 공개 자료 범위에 맞췄습니다.
  • 상권 경계는 손으로 그렸고, 3D는 상자 모델, 거리뷰는 그림, 평면은 개략 배치도입니다.
  • 적합도 · 경쟁 강도 · 간판 노출도 점수 공식은 아직 검증되지 않았습니다.
  • CDN(unpkg · jsDelivr · OpenFreeMap)이 막힌 곳에서는 지도 · 3D가 보이지 않습니다.
  • 가계부는 브라우저에만 저장되고, 다른 기기와 동기화되지 않습니다.
  • 원본 요구사항(바탕화면 압축폴더)과 placeos.co.kr 실제 화면은 작업 환경에서 열 수 없어 직접 대조하지 못했습니다.

  • P1 환산보증금(보증금 + 월세×100, 서울 9억)으로 상가임대차 보호 범위 자동 판정
  • P1 다중이용업소(100㎡, 지하 66㎡) · 근생 1종/2종 · 구역 규제 자동 안내
  • P1 카드 · 배달 수수료 자동 계산, 업종별 폐업 위험 표시
  • P2 서울시 상권분석 OpenAPI · 소상공인 상가정보 · LOCALDATA · 공정위 가맹 API 연동
  • P2 보행 거리 · 층을 반영한 경쟁 분석, 실측 공간 데이터(스캔 · 도면)
  • P3 행사 · 지원사업 자동 매칭, 방문 데이터 기반 분석

용어 풀이

바이브 코딩원하는 결과를 말로 설명하면 AI가 코드를 써 주는 개발 방식. 이 문서의 '프롬프트'가 그 설명입니다.
CDN라이브러리 파일을 인터넷에서 바로 불러오는 서버(unpkg, jsDelivr). 막히면 지도 · 3D가 안 뜹니다.
integrity (SRI)불러온 파일이 원본과 같은지 브라우저가 확인하는 값. 버전을 바꾸면 값이 달라져 로딩이 막힙니다.
WebGL브라우저에서 3D를 그리는 기능. 하드웨어 가속이 꺼져 있으면 안 됩니다.
localStorage브라우저에 작은 데이터를 저장하는 곳. 가계부 입력값이 여기에 저장됩니다.
SHA-256 해시파일 지문. 두 파일의 해시가 같으면 내용이 완전히 같습니다.
브랜치 · 커밋 · 푸시작업을 따로 보관하는 가지 · 저장 시점 · GitHub에 올리기.
